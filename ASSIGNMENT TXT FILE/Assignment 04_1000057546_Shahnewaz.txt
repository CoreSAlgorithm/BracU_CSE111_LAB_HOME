public class Triangle
{
  public int a = 0;
  public int b = 0;
  public int c = 0;
  public int perimeter = 0;
  public Triangle(int a,int b,int c)
  {
    this.a = a;
    this.b = b;
    this.c = c;
    this.perimeter = (a+b+c);
  }
  public void triangleDetails()
  {
    System.out.println("Three sides of the triangle are: "+a+", "+b+", "+c);
    System.out.println("Perimeter: "+this.perimeter);
  }
  public String printTriangleType()
  {
    if(this.a == this.b && this.b == this.c)
    {
      return "This is an Equilateral Triangle.";
    }
    else if(this.a != this.b && this.b != this.c && this.c != this.a)
    {
      return "This is an Scalene Triangle.";
    }
    else
    {
      return "This is an Isosceles Triangle.";
    }
  }
  public void compareTriangles(Triangle t)
  {
    if (this == t)
    {
        System.out.println("These two triangle objects have the same address.");
    }
    else if (this.a == t.a && this.b == t.b && this.c == t.c)
    {
        System.out.println("Addresses are different but the sides of the triangles are equal.");
    }
    else if (this.a != t.a && this.b != t.b && this.c != t.c ||
             this.a == t.a && this.b != t.b && this.c != t.c ||
             this.a != t.a && this.b == t.b && this.c != t.c ||
             this.a != t.a && this.b != t.b && this.c == t.c && this.perimeter != t.perimeter)
    {
        System.out.println("Addresses, length of the sides and perimeter all are different.");
    }
    else if (this.perimeter == t.perimeter)
    {
        System.out.println("Only the perimeter of both triangles is equal.");
    }
  }
}

public class Player
{
  public String name = "";
  public int hp = 100;  
  public String enemy = "";
  public String[] defeated = new String[5];
  public int count = 0;
  public Player(String name, int hp)
  {
    this.name = name;
    this.hp = hp;
    System.out.println(this.name+" joined the game");
    System.out.println("Hp: "+this.hp);
  }
  public Player(String name)
  {
    this.name = name;
    System.out.println(this.name+" joined the game");
    System.out.println("Hp: "+this.hp);
  }
  public void viewInfo()
  {
    System.out.println("Player name: "+this.name);
    System.out.println("Current Hp: "+this.hp);
    System.out.println("Defeated");
    for(int i=0; count>i; i++)
    {
      System.out.print(this.defeated[i]+", ");
    }
    System.out.print("\n ");
  }
  public void defeatVillain(String enemy, int hp)
  {
    this.enemy = enemy;
    if(this.hp < hp)
    {
      System.out.println("failed to defeat "+this.enemy);
    }
    else if(this.hp == hp)
    {
      this.defeated[count] = enemy;
      this.hp += hp;
      System.out.println("Defeated "+this.enemy);
      count++;
    }
  }
  public void defeatVillain(int a, String enemy, int hp)
  {
    this.hp = (a*this.hp);
    System.out.println("HP with "+a+"x boost: "+this.hp);
    if(this.hp > hp)
    {
      this.enemy = enemy;
      this.defeated[count] = enemy;
      this.hp += hp;
      System.out.println("Defeated "+this.enemy);
      count++;
    }
  }
  public void defeatVillain(Player p)
  {
    if(this.hp > p.hp)
    {     
      this.defeated[count] = p.name;
      this.hp += p.hp;
      System.out.println("Defeated "+p.name);
      count++;
    }
    else if(this.hp < p.hp)
    {
      System.out.println("failed to defeat "+p.name);
    }
  }
}

public class Spaceship
{
  public String name = "";
  public int totalCapacity = 0;  
  public int capacity = 0;
  public String[] metarial = new String[100];
  public int count = 0;
  public Spaceship(String name, int totalCapacity)
  {
    this.name = name;
    this.totalCapacity = totalCapacity;
  }
  public void loadCargo(Cargo metarial)  
  {  
    this.capacity += metarial.getWeight();
    if(this.totalCapacity >= capacity)
    {
      this.metarial[count] = metarial.getName();
      count++;
    }
    else
    {
      System.out.println("Warning: Unable to load "+metarial.getName()+" inside "+this.name+". Exceeds capacity by "+(this.capacity-this.totalCapacity)+" .");
    }
  }
  public void displayDetails()
  {
    System.out.println("Spaceship Name: "+this.name);
    System.out.println("Capacity: "+this.totalCapacity);
    System.out.println("Current cargo weight :"+this.capacity);
    System.out.print("Cargo: ");
    for(int i=0; i<count; i++)
    {
      System.out.print(this.metarial[i]+" ");
    }
    System.out.println();
  }
}

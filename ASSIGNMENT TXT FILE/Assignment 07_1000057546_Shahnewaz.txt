public class Concert
{
  public String zone = "";
  public int quantity = 0;
  public static int sold = 0;
  public static int vipsold = 0;
  public String venue = "";
  public String showtime = "";
  public String[] artist = new String[5];
  public int count = 0;
  public Concert(String venue, String showtime)
  {
    this.venue = venue;
    this.showtime = showtime;
  }
  public void addArtist(String artist)
  {
    this.artist[count] = artist;
    count++;
  }
  public void showDetails()
  {
    System.out.println("Vanue: "+this.venue);
    System.out.println("Showtime: "+this.showtime);
    System.out.println("Artist: ");
    if(count > 0)
    {
      for(int i=0; i<count; i++)
      {
        System.out.println(artist[i]);
      }
    }
    System.out.println("Tickets sold in this concert: "+this.sold);
  }
  public void buyTicket(String zone, int quantity)
  {
    this.zone = zone;
    this.quantity = quantity;
    this.sold += quantity;
    if(zone.equals("A"))
    {
      System.out.println("Total Price: "+(500*quantity));
    }
    else if(zone.equals("B"))
    {
      System.out.println("Total Price: "+(1000*quantity));
    }
    if(zone.equals("VIP"))
    {
      System.out.println("Total Price: "+(2000*quantity));
    }
  }
  public void buyTicket(String zone)
  {
    this.buyTicket(zone,1);
  }
  public void buyTicket()
  {
    this.vipsold += 1;
    System.out.println("Total Price: 2000");
  }
  public static void showTotalTicketsSold()
  {
    System.out.println("Total tickets sold (all venues): "+(sold+vipsold));
  }
}

public class BikeDriver extends GenericDriver
{
  public static String[] restrictedAreas = new String[3];
  public static int count = 0;
  public String profile = null;
  public BikeDriver(String name, String profile)
  {
    super(name);
    this.profile = profile;
  }
  public String toString()
  {
    if(profile == null)
    {
      return (getName()+"'s driver profile is Not Premium");
    }
    else
    {
      return (getName()+"'s driver profile is Premium");
    }
  }
  public static void restrictedAreas(String areas)
  {
    restrictedAreas[count] = areas;
    count++;
  }
  public void fightRestriction(String[] areas)
  {
    for (String area : areas)
    {
      boolean found = false;
      for (int i = 0; i < count; i++)
      {
        if (area.equals(restrictedAreas[i]))
        {
          found = true;
          break;
        }
      }
      if (found)
      {
        System.out.println(getName() + " cannot enter " + area);
      }
      else
      {
        System.out.println(getName() + " can enter " + area);
      }
    }
  }
}

public class Car extends Vehicle
{
  public static Car[] car = new Car[4];
  public int seat = 0;
  public static int count = 0;
  public static int totalcar = 0;
  public String status = "Available";
  public static boolean sold1 ;
  public Car(String model, int price,int seat)
  {
    super(model, price);
    car[count] = this;
    this.seat = seat;
    vehicleId = ("CAR00"+(count+1));
    count++;
    totalcar++;
    System.out.println("Vehicle ID: "+vehicleId+" created");
  }
  public void vehicleDetail()
  {
    System.out.println("Model: " + model + ", Price: $" + price);
    System.out.print("Status: ");
    if(sold)
    {
      System.out.print("Sold\n");
    }
    else
    {
      System.out.print("Available\n");
    }
    System.out.println("Type: Regular Car");
    System.out.println("Seat: "+seat);
  }
  public static void showAllAvailableCars()
  {
    System.out.println("Total Car: "+totalcar);
    System.out.println("Available car: ");
    for(int i=0; i<count; i++)
    {
      if (car[i].status.equals("available"))
      {
        System.out.println(car[i].vehicleId + " : " + car[i].model);
      }
    }
  }
  public static void markAsSold(Car c)
  {
    sold1 = true;
    c.status = "sold";
  }
  public static void showAllCars()
  {
    System.out.println("Total Car: "+totalcar);
    for(int i=0; i<count; i++)
    {
      System.out.println(car[i].vehicleId+" : "+car[i].model+" - "+car[i].status);
    }
  }
}

Manager Class-------

public class Manager extends Employee
{
    private double bonus;
    private double finalSalary;

    public Manager(String name, double baseSalary, int hoursWorked, double bonus) {
        super(name, baseSalary, hoursWorked);
        this.bonus = bonus;
    }

    public void calculateSalary() {
        finalSalary = getBaseSalary() + (getBaseSalary() * bonus / 100);
    }

    public void requestIncrement(double amount) {
        if (getHoursWorked() > 80) {
            setBaseSalary(getBaseSalary() + amount/2);
            System.out.println("$" + amount/2 + " Increment approved.");
        } else {
            System.out.println("Increment denied.");
        }
    }

    public void displayInfo() {
        super.displayInfo();
        System.out.println("Bonus: " + bonus + " %");
        System.out.println("Final Salary: $" + finalSalary);
    }
}

public class Developer extends Employee {
    private String language;
    private double finalSalary;

    public Developer(String name, double baseSalary, int hoursWorked, String language) {
        super(name, baseSalary, hoursWorked);
        this.language = language;
    }

    public void calculateSalary() {
        if (language.equals("Java")) {
            finalSalary = getBaseSalary() + 700;
        } else {
            finalSalary = getBaseSalary();
        }
    }

    public void displayInfo() {
        super.displayInfo();
        System.out.println("Language: " + language);
        System.out.println("Final Salary: $" + finalSalary);
    }
}

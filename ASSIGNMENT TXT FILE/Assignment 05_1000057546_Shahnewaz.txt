public class Product
{
  public static String[] name = new String[3];
  public static int count = 0;
  public static int[] qty = new int[3]; 
  public Product(String name, int qty)
  {
    if(count < 3)
    {
      this.name[count] = name;
      this.qty[count] = qty;
      count++;
      System.out.println("Stored: "+name);
    }
    else
    {
      System.out.println("Storage is full! Cannot add.");
    }
  }
  public static void displayProducts()
  {
    System.out.println("=== Stored Products ===");
    System.out.println(name[0]+" - QTY: "+qty[0]);
    System.out.println(name[1]+" - QTY: "+qty[1]);
    System.out.println(name[2]+" - QTY: "+qty[2]);
  }
  public static void buy(String nam, int qt)
  {
    for(int i=0; i<count; i++)
    {
      if(name[i].equals(nam))
      {
        if(qty[i] > qt)
        {
          qty[i] -= qt;
          System.out.println("Product sold");
          return;
        }
        else
        {
          System.out.println("Quantity low");
          return;
        }
      }
    }
    System.out.println("Product not found");
  }
  
}

public class Character
{
  public String name = "";
  public static int charecter = 0;
  public static int kids = 0;
  public static int teens = 0;
  public static int adults = 0;
  public static int id = 0;
  public String group = "Teen";
  public int health = 0;
  public static double total = 0;
  public static String strong = "";
  public static int sthealth = 0;
  public Character(String name, String group, int health)
  {
    this.name = name;
    charecter++;
    if(group.equals("Kid"))
    {
      this.group = group;
      kids++;
    }
    else if(group.equals("Adult"))
    {
      this.group = group;
      adults++;
    }
    else if(group.equals("Teen"))
    {
      this.group = group;
      teens++;
    }
    this.health = health;
    total += health;
    if(sthealth < health)
    {
      strong = name;
      sthealth = health;
    }
    Character.id++;
  }
  public Character(String name, int health)
  {
    this.name = name;
    charecter++;
    this.health = health;
    total += health;
    teens++;
    if(sthealth < health)
    {
      strong = name;
      sthealth = health;
    }
    ++id;
  }
  public void printDetails()
  {
    System.out.println("ID: "+id);
    System.out.println("Name: "+this.name);
    System.out.println("Group: "+this.group);
    System.out.println("Health: "+this.health);
  }
  public static void printStats()
  {
    System.out.println("Total Character: "+charecter);
    System.out.println("Kids: "+kids);
    System.out.println("Teens: "+teens);
    System.out.println("Adult: "+adults);
    System.out.println("Average Health: "+total/id);
    if(strong == "")
    {
      System.out.println("Strongest Character: None");
    }
    else
    {
      System.out.println("Strongest Charecter: "+strong+"(Health"+sthealth+")");
    }
  }
}

public class Artifact
{
  private String name = "";
  private String owner = "Okabe";
  private int power = 0;
  public static Artifact[] vault = new Artifact[4];
  public static int count = 0;
  public Artifact(String name, String owner)
  {
    this.name = name;
    this.owner = owner;
    this.power = CalcPower();
  }
  public Artifact(String name)
  {
    this.name = name;
  }
  public String GetName()
  {
    return name;
  }
  public String Getowner()
  {
    return owner;
  }
  public int CalcPower()
  {
    int power = 0;
    int n = name.length();
      if(n%2 == 0)
      {
        for(int j=0; j<n; j+=2)
        {
          power += (int)name.charAt(j);
        }
      }
      else
      {
        for(int j=1; j<n; j+=2)
        {
          power += (int)name.charAt(j);
        }
      }
    return power;
  }
  public static void AddtoVault(Artifact obj)
  {
    if(count < 4)
    {
      Artifact.vault[count] = obj;
      System.out.println(obj.owner+" added "+obj.name+" successfully to the vault.");
      count++;
    }
    else
    {
      System.out.println(obj.owner+" unsuccessful in adding artifact to the vault!!");
    }
  }
  public static void labReport()
  {
    System.out.println("=== Future Gadget Lab ===");
    for(int i=0; i<count; i++)
    {
      System.out.println(vault[i].name+" added by "+vault[i].owner+" has power of "+vault[i].power);
    }
  }
  public static String strongest()
  {
    if(count == 0)
    {
      return "No artifacts";
    }
    Artifact max = vault[0];
    for(int i=1; i<count; i++)
    {
      if(vault[i].power > max.power)
      {
        max = vault[i];
      }
    }
    return max.name;
  }
  public void revealArtifact()
  {
    System.out.println(name + " added by " + owner + " has power of " + power + ".");
  }
  public void changeName(String newName)
  {
    this.name = newName;
    this.power = CalcPower();
    System.out.println("Name changed and power recalculated.");
  }
}

public class Animalkeepers
{
  public String keepername = "";
  private  static int keeperId = 101;
  public static int totalkeeper = 0;
  public static String[] animal = new String[5];
  public static String[] work = new String[5];
  public static String[] assignedKeeper = new String[5];
  public static int count = 0;
  public static int taskassign = 0;
  public static String[] Animals = {"Lion", "Tiger", "Seal", "Gorilla", "Deer"};
  public Animalkeepers(String keepername)
  {
    this.keepername = keepername;
    System.out.println(this.keepername+" with ID "+keeperId+" got the job!");
    keeperId++;
    totalkeeper++;
  }
  public static void details()
  {
    if(totalkeeper == 0)
    {
      System.out.println("No Animal Keepers working yet.");
    }
    else
    {
      System.out.println("Total Animal Keeper: "+totalkeeper);
      System.out.println("Total Task assigned: "+taskassign);
      printTasks();
    }
  }
  public static void printTasks()
  {
    if(taskassign == 0)
    {
      System.out.println("No tasks assigned.");
      return;
    }
    for(int i=0; i<count; i++)
    {
      System.out.println(work[i]+" (keeper - "+assignedKeeper[i]+") === "+animal[i]);
    }
  }
  public void doTask(String animalname, String workname)
  {
    boolean validanimal = false;
    for(int i=0; i<Animals.length; i++)
    {
      if(animalname.equals(Animals[i]))
      {
        validanimal = true;
        break;
      }
    }
    if(!validanimal)
    {
      System.out.println("Animal not in the Safari.");
    }
    for(int i=0; i<count; i++)
    {
      if(animal[i].equals(animalname))
      {
        work[i] = workname;
        assignedKeeper[i] = this.keepername;
        System.out.println("Task assigned to " + this.keepername);
        return;
      }
    }
     animal[count] = animalname;
     work[count] = workname;
     assignedKeeper[count] = this.keepername;
     count++;
     taskassign++;
     System.out.println("Task assigned to " + this.keepername);
  }
}

public class Event
{
  public static String[] eventname = new String[5];
  public static String[] eventdate = new String[5];
  private String event = "";
  private String date = "";
  public static int count = 0;
  public Event(String event, String date)
  {
    if(count < 5)
    {
      this.event = event;
      this.date = date;
      eventname[count] = event;
      eventdate[count] = date;
      count++;
    }
    else
    {
      System.out.println("Maximum event.");
    }
  }
  public String details()
  {
    return "Name: " + this.event + "\nDate: " + this.date;
  }
  public String getEventname()
  {
    return this.event;
  }
  public String getEventdate()
  {
    return this.date;
  }
  public static void allEventInfo()
  {
    if(count == 0)
    {
      System.out.println("Total events: "+count);
      System.out.println("Event Details: ");
    }
    else
    {
      System.out.println("Total events: "+count);
      System.out.println("Event Details: ");
      for(int i=1; i<=count; i++)
      {
        System.out.println("Event +"+i+": ");
        System.out.println("Name: "+eventname[i-1]);
        System.out.println("Date: "+eventdate[i-1]);
      }
    }
  }
}

  {
    for(int i=0; i<count; i++)
